% Number check used by the converter after every numbered heading and
% every table caption.  #1 = LaTeX counter, #2 = number written in source.
\newcommand\ThesisCheckNumber[2]{%
  \edef\ThesisGotNumber{\csname the#1\endcsname}%
  \edef\ThesisWantNumber{#2}%
  \ifx\ThesisGotNumber\ThesisWantNumber\else
    \PackageError{thesis}{Number mismatch for #1: the Markdown source
      says #2 but LaTeX produced \ThesisGotNumber}{Fix the source numbering
      or the chapter counter in the driver; do not suppress this error.}%
  \fi}

% Caption emitted by the converter from a "**Table N.M --- Title**" line.
%
% Keeping captions, tables and notes together (added in the ch05 session; the
% ch05 preview is the first unit with tables directly under headings, tables
% of 10-20 rows and table notes). Each pass writes two measurements to the
% .aux file: the height of every longtable (header + body + foot; the body is
% one longtable chunk for tables of up to 20 rows) and the height of any note
% that follows it. On the next pass:
%  - the caption requests room (\Needspace) for caption + table + note, at
%    least 8 lines, so a table that fits on a page is not split and its
%    caption is not separated from it;
%  - a heading directly followed by a caption requests room for heading +
%    caption + table + note, at least 12 lines, so the page break falls before
%    the heading rather than between heading and table;
%  - a table followed by a note ends with \penalty10000 instead of
%    longtable's \penalty0, so the note stays on the page of the last row.
% Tables taller than 0.8\textheight get no extra keep and break normally
% (longtable repeats the column heads). Tables without notes and headings not
% followed by a table behave as before, apart from the caption's \Needspace
% (\needspace misjudged the room directly after a heading; found in ch05).
\makeatletter
\newcount\ThesisHeadingNo
\newdimen\ThesisNeed
\newsavebox\ThesisNoteBox
% A dummy counter solely so that \LTcaptype can safely name something that
% \newcounter has actually defined. pandoc's longtable* output (no \caption
% call) sets \LTcaptype{none} on the assumption that an uncaptioned
% longtable* never touches any counter; a newer LaTeX kernel's PDF-tagging
% hooks inside \endlongtable call \refstepcounter regardless, which
% ltcaption redirects through \LTcaptype -- crashing on the undefined
% counter "none". Redirecting it here to this always-defined, never-read
% counter instead keeps that redirection harmless without affecting the
% real \c@table counter that \ThesisTableCaption's own \captionof{table}
% call increments (build.py's converter rewrites \LTcaptype{none} to this
% counter's name for every generated table).
\newcounter{ThesisDummyLTCaptype}
\newcommand\ThesisLTHeight[2]{\global\@namedef{ThesisLTHeight@#1}{#2}}
\newcommand\ThesisNoteHeight[2]{\global\@namedef{ThesisNoteHeight@#1}{#2}}

\newcommand\ThesisHeadingWithTable[2]{\global\@namedef{ThesisHeadingWithTable@#1}{#2}}
% \ThesisNeed := recorded height of table #1 plus its note (0 if unknown or tall)
\newcommand\ThesisSetNeed[1]{%
  \global\ThesisNeed\z@
  \ifcsname ThesisLTHeight@#1\endcsname
    \global\advance\ThesisNeed\csname ThesisLTHeight@#1\endcsname\relax\fi
  \ifcsname ThesisNoteHeight@#1\endcsname
    \global\advance\ThesisNeed\csname ThesisNoteHeight@#1\endcsname\relax
    \global\advance\ThesisNeed 9.75bp\fi
  \ifdim\ThesisNeed>.8\textheight\global\ThesisNeed\z@\fi}
\newcommand\ThesisHeadingNeedspace{%
  \global\advance\ThesisHeadingNo\@ne
  \ifcsname ThesisHeadingWithTable@\the\ThesisHeadingNo\endcsname
    \ThesisSetNeed{\csname ThesisHeadingWithTable@\the\ThesisHeadingNo\endcsname}%
    \ifdim\ThesisNeed>\z@\global\advance\ThesisNeed 7\baselineskip\fi
    % Same unavailable-height situation as \ThesisTableCaption above:
    % \ThesisNeed is always 0 here this build, so the blind fallback
    % below is what actually governs every heading immediately followed
    % by a table. The previous 12\baselineskip (~3.25in) blindly
    % reserved before every such heading (9 of 9 Chapter 5 sections,
    % plus 3 appendix sections) forced premature page breaks; 4\baselineskip
    % still left avoidable blank space at several page bottoms once the
    % main-text page budget became tight (R-page-limit, 2026-10-17).
    % 2\baselineskip (~1 line plus inter-paragraph space) is still enough
    % to keep the heading from being orphaned alone at a page bottom,
    % without reserving room for table content this mechanism cannot see.
    \ifdim\ThesisNeed<2\baselineskip\global\ThesisNeed 2\baselineskip\fi
    \par\Needspace{\ThesisNeed}\fi}
\pretocmd{\section}{\ThesisHeadingNeedspace}{}{\PackageError{thesis-preamble}{could not patch \string\section}{}}
\pretocmd{\subsection}{\ThesisHeadingNeedspace}{}{\PackageError{thesis-preamble}{could not patch \string\subsection}{}}
\pretocmd{\subsubsection}{\ThesisHeadingNeedspace}{}{\PackageError{thesis-preamble}{could not patch \string\subsubsection}{}}
% record the table height just before longtable outputs its last chunk; end
% with \penalty10000 when a note follows this table
\patchcmd{\endlongtable}{\LT@echunk\LT@start}
  {\LT@echunk
   \if@filesw\immediate\write\@auxout{\string\ThesisLTHeight{\the\c@LT@tables}{\the\dimexpr
     \ht\z@+\dp\z@+\ht\LT@head+\dp\LT@head+\ht\LT@lastfoot+\dp\LT@lastfoot+\LTpre\relax}}\fi
   \LT@start}{}%
  {\PackageWarning{thesis-preamble}{could not patch \string\endlongtable\space(height) -- installed longtable's internals differ from this template's assumption; falling back to no table-height tracking (tables/captions/notes may occasionally break across a page boundary where the keep-together logic would otherwise have prevented it; nothing else is affected)}}
\patchcmd{\endlongtable}{\endgraf\penalty\z@\addvspace\LTpost}
  {\endgraf\ifcsname ThesisNoteAfterLT@\the\c@LT@tables\endcsname
     \penalty\@M\else\penalty\z@\fi\addvspace\LTpost}{}%
  {\PackageWarning{thesis-preamble}{could not patch \string\endlongtable\space(penalty) -- installed longtable's internals differ from this template's assumption; falling back to longtable's own default penalty (a table's note may occasionally separate from its table across a page boundary; nothing else is affected)}}
\newenvironment{ThesisCaptionBlock}{\par}{\par}
\newcommand\ThesisTableCaption[2]{%
  \ThesisSetNeed{\the\numexpr\c@LT@tables+1\relax}%
  \if@nobreak\if@filesw\immediate\write\@auxout{\string\ThesisHeadingWithTable
    {\the\ThesisHeadingNo}{\the\numexpr\c@LT@tables+1\relax}}\fi\fi
  \ifdim\ThesisNeed>\z@\global\advance\ThesisNeed 3\baselineskip\fi
  % The real per-table height is unavailable this build (the \endlongtable
  % height-tracking patch does not match this installed longtable's
  % internals -- see the \ThesisDummyLTCaptype note above -- so
  % \ThesisNeed is always 0 here). The previous blind fallback of
  % 8\baselineskip (~2.2in) reserved that much room before literally
  % every table regardless of its actual size, forcing premature page
  % breaks across this thesis's 12 tables at final assembly (54->48
  % pages before this change, together with other edits). 3\baselineskip
  % was enough to keep a caption from being orphaned alone at a page
  % bottom -- the actual failure mode this mechanism exists to prevent
  % -- without blindly reserving two-plus inches for every table.
  % 2\baselineskip tightens this further once the main-text page budget
  % became tight (R-page-limit, 2026-10-17): still enough for the
  % caption's own heading line, not enough to force a break the actual
  % content would not have needed.
  \ifdim\ThesisNeed<2\baselineskip\global\ThesisNeed 2\baselineskip\fi
  \par\Needspace{\ThesisNeed}\addvspace{9.75bp}%
  \begin{ThesisCaptionBlock}\captionsetup{hypcap=false}%
    \captionof{table}{#2}\label{tab:#1}\ThesisCheckNumber{table}{#1}%
  \end{ThesisCaptionBlock}\nopagebreak}

% Figure captions (added 2026-10-17, R-page-limit figure-caption fix): used
% INSIDE a \begin{figure}[H] ... \end{figure} environment, immediately after
% \includegraphics, so the real \caption call and the \thesisfigurecaption
% font setting above both attach to the actual float (unlike the table
% mechanism above, a figure this size needs no needspace reservation -- [H]
% already pins it at this point in the text). #1 = number written in the
% Markdown source (checked against the real counter); #2 = caption text.
\newcommand\ThesisFigureCaption[2]{%
  \captionsetup{hypcap=false}%
  \caption{#2}\label{fig:#1}\ThesisCheckNumber{figure}{#1}}
